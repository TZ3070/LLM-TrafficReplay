(async () => {
  const metric =
    'vllm:prefix_cache_queries_total{model_name="Qwen3.8-2"}';

  const checks = {
    '该模型最近1分钟样本数':
      `count_over_time(${metric}[1m])`,
    '该模型最近5分钟样本数':
      `count_over_time(${metric}[5m])`,
    '所有模型最近1分钟样本数及标签':
      'count_over_time(vllm:prefix_cache_queries_total[1m])'
  };

  const results = {};
  for (const [name, query] of Object.entries(checks)) {
    const params = new URLSearchParams({
      query,
      start: '1792066643',
      end: '1792066643',
      step: '15'
    });
    const response = await fetch(
      '/api/prometheus/query_range?' + params
    );
    results[name] = await response.json();
  }

  console.log(JSON.stringify(results, null, 2));
})();
